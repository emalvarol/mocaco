Convergence analysis for Monte Carlo series using Polars DataFrames for fastest Python performance.
The library provides a simple API for applying different convergence and stopping criteria to Monte Carlo simulation results.

## Quickstart

```python
import polars as pl
import mocaco as mcc

# Prepare your Monte Carlo samples
df = pl.DataFrame({"it": range(1000), "value": your_samples})

# Create a normalized sample frame
samples = mcc.samples(df, it_col="it", target_col="value")

# Run a convergence criterion
result = mcc.convergence.clt_uni_abs(samples, threshold=0.01)

# Inspect results
print(result.is_converged)  # True/False
print(result.estimate)  # Final cumulative mean
print(result.error)  # Final CLT-based absolute error
print(result.n)  # Number of samples
# and more...
```

## API Overview

| Function | Description |
|----------|-------------|
| `mcc.samples(df, it_col, target_col)` | Create a normalized sample frame from a Polars DataFrame |
| `mcc.convergence(samples, method="...")` | Generic invocation of any registered criterion |
| `mcc.convergence.clt_uni_abs(samples, ...)` | Typed invocation of a specific criterion |
| `mcc.methods()` | List all registered convergence criteria |
| `mcc.describe("clt_uni_abs")` | Get metadata and parameter info for a criterion |

## Available Methods

- [`clt_uni_abs`] — CLT-based absolute error threshold

from __future__ import annotations

import time
from dataclasses import replace
from typing import TYPE_CHECKING, Any

from . import methods as _methods  # noqa: F401
from .protocols import SampleFrame
from .registry import registry
from .result import ConvergenceResult, GroupedConvergenceResult

if TYPE_CHECKING:
    from collections.abc import Callable

    import polars as pl

    from .protocols import Criterion


def samples(
    df: pl.DataFrame,
    *,
    target_col: str,
    it_col: str | None = None,
    group_cols: str | list[str] | None = None,
) -> SampleFrame:
    
    return SampleFrame(df=df, target_col=target_col, it_col=it_col, group_cols=group_cols)


class Convergence:
    

    def _execute_criterion(
        self,
        criterion: Criterion,
        samples: SampleFrame,
        kwargs: dict[str, Any],
    ) -> ConvergenceResult | GroupedConvergenceResult:
        start_time = time.perf_counter()

        params = criterion.params_type(**kwargs)
        result = criterion.run(samples=samples, params=params)

        elapsed = time.perf_counter() - start_time

        if isinstance(result, dict):
            n_groups = len(result)
            per_group_elapsed = elapsed / n_groups if n_groups > 0 else elapsed

            grouped_dict = GroupedConvergenceResult(
                {
                    group_key: replace(res, execution_time_sec=per_group_elapsed)
                    for group_key, res in result.items()
                }
            )
            return grouped_dict

        return replace(result, execution_time_sec=elapsed)

    def __call__(
        self,
        samples: SampleFrame,
        *,
        method: str,
        **kwargs: Any,
    ) -> ConvergenceResult | GroupedConvergenceResult:
        
        criterion = registry.get(method)
        return self._execute_criterion(criterion, samples, kwargs)

    def __getattr__(self, name: str) -> Callable[..., ConvergenceResult | GroupedConvergenceResult]:
        
        criterion = registry.get(name)

        def _method(
            samples: SampleFrame, **kwargs: Any
        ) -> ConvergenceResult | GroupedConvergenceResult:
            return self._execute_criterion(criterion, samples, kwargs)

        return _method


convergence = Convergence()


def methods() -> tuple[str, ...]:
    
    return registry.names()


def describe(name: str) -> None:
    
    return registry.describe(name)

import logging

import polars as pl

logger = logging.getLogger(__name__)


@dataclass(frozen=True)
class SampleFrame:
    

    df: pl.DataFrame
    target_col: str
    it_col: str | None = None
    group_cols: str | list[str] | None = None

    def __post_init__(self) -> None:
        
        # Normalize group_cols to a list[str]
        if isinstance(self.group_cols, str):
            object.__setattr__(self, "group_cols", [self.group_cols])
        elif self.group_cols is None:
            object.__setattr__(self, "group_cols", [])

        # Resolve and normalize it_col name
        if self.it_col is None:
            it_col_name = "it"
            if self.group_cols:
                new_df = self.df.with_columns(
                    pl.int_range(1, pl.len() + 1).over(self.group_cols).alias(it_col_name)
                )
            else:
                new_df = self.df.with_columns(pl.int_range(1, pl.len() + 1).alias(it_col_name))
            object.__setattr__(self, "df", new_df)
            object.__setattr__(self, "it_col", it_col_name)
            logger.info(
                f"it_col was None — auto-created column '{it_col_name}' with sequential indices."
            )
        else:
            it_col_name = self.it_col

        # Validate structural integrity (it_col_name is guaranteed to be str here)
        missing = [col for col in (it_col_name, self.target_col) if col not in self.df.columns]
        if missing:
            raise ValueError(
                f"Columns not found in DataFrame: {missing}. Available columns: {self.df.columns}"
            )

        if self.df.is_empty():
            raise ValueError("The sample DataFrame cannot be empty.")

        # Validate Data Types (it_col and target_col must be numeric)
        it_dtype = self.df.get_column(it_col_name).dtype
        if not it_dtype.is_numeric():
            raise TypeError(
                f"Iteration column '{it_col_name}' must be numeric (int or float), got {it_dtype}."
            )

        target = self.df.get_column(self.target_col)
        if not target.dtype.is_numeric():
            raise TypeError(
                f"Target column '{self.target_col}' must be numeric (int or float), got {target.dtype}."
            )

        assert isinstance(self.group_cols, list)
        for col in self.group_cols:
            dtype = self.df.get_column(col).dtype
            if (
                not isinstance(
                    dtype,
                    (
                        pl.Int8,
                        pl.Int16,
                        pl.Int32,
                        pl.Int64,
                        pl.UInt8,
                        pl.UInt16,
                        pl.UInt32,
                        pl.UInt64,
                        pl.Float32,
                        pl.Float64,
                        pl.String,
                        pl.Categorical,
                        pl.Enum,
                        pl.Boolean,
                        pl.Date,
                        pl.Datetime,
                    ),
                )
                and not dtype.is_numeric()
            ):
                raise TypeError(
                    f"Group column '{col}' has unsupported data type {dtype}. "
                    "Supported types are numeric, string, categorical, enum, boolean, date, and datetime."
                )

        # Input validation and edge case handling (Null, NaN, Inf)
        if target.is_null().any():
            raise ValueError(f"Target column '{self.target_col}' contains Null (missing) values.")

        if target.dtype in (pl.Float32, pl.Float64):
            if target.is_nan().any():
                raise ValueError(f"Target column '{self.target_col}' contains NaN values.")
            if target.is_infinite().any():
                raise ValueError(
                    f"Target column '{self.target_col}' contains Infinite (Inf) values."
                )

from __future__ import annotations

from typing import TYPE_CHECKING, Any, TypeVar, overload

from rich.console import Console
from rich.panel import Panel
from rich.table import Table

if TYPE_CHECKING:
    from collections.abc import Callable

    from .protocols import Criterion

C = TypeVar("C", bound=type)

console = Console()


class CriterionRegistry:
    def __init__(self) -> None:
        self._criteria: dict[str, Criterion] = {}

    # Overload 1: Decorator usage (no criterion provided)
    @overload
    def register(self, criterion: None = None, *, overwrite: bool = False) -> Callable[[C], C]: ...

    # Overload 2: Programmatic usage (criterion instance provided)
    @overload
    def register(self, criterion: Criterion, *, overwrite: bool = False) -> Criterion: ...

    def register(
        self,
        criterion: Criterion | None = None,
        *,
        overwrite: bool = False,
    ) -> Any:
        # 1. Programmatic use: registry.register(MyCriterion())
        if criterion is not None:
            name = criterion.name

            if not name:
                raise ValueError("A criterion must define a non-empty name.")

            if name in self._criteria and not overwrite:
                raise ValueError(f"Criterion {name!r} is already registered.")

            self._criteria[name] = criterion
            return criterion

        # 2. Decorator use: @registry.register()
        def wrapper(cls: C) -> C:
            instance = cls()
            self.register(instance, overwrite=overwrite)
            return cls

        return wrapper

    def get(self, name: str) -> Criterion:
        try:
            return self._criteria[name]
        except KeyError as exc:
            available = ", ".join(self.names())
            raise ValueError(
                f"Unknown convergence criterion: {name!r}. Available criteria: {available}"
            ) from exc

    def names(self) -> tuple[str, ...]:
        return tuple(self._criteria.keys())

    def describe(self, name: str) -> None:
        # 1. Fetch the criterion from the registry
        try:
            criterion = registry.get(name)
        except ValueError:
            console.print(f"[bold red]Error:[/bold red] Method '{name}' not found in registry.")
            return

        # 2. Print the Method Name and Main Description in a Panel
        console.print()
        console.print(
            Panel(
                f"{criterion.description}",
                title=f"Method: [bold cyan]{name}[/bold cyan]",
                title_align="left",
                border_style="cyan",
                expand=False,
            )
        )

        # 3. Create a Table for the Parameters
        table = Table(
            title="Parameters",
            title_style="bold magenta",
            title_justify="left",
            show_header=True,
            header_style="bold black on white",
        )

        table.add_column("Name", style="bold cyan", no_wrap=True)
        table.add_column("Type", style="green")
        table.add_column("Required", justify="center")
        table.add_column("Default", style="yellow")
        table.add_column("Description", style="dim")

        # 4. Introspect the Pydantic V2 model fields
        params_type = criterion.params_type

        for field_name, field_info in params_type.model_fields.items():
            # Get type name cleanly
            field_type = getattr(field_info.annotation, "__name__", str(field_info.annotation))

            is_required = field_info.is_required()
            req_icon = "[green]✔[/green]" if is_required else "[red]✘[/red]"
            default_val = "-" if is_required else repr(field_info.default)
            desc = field_info.description or "No description provided."

            table.add_row(field_name, field_type, req_icon, default_val, desc)

        # 5. Render the table
        console.print(table)
        console.print()

from __future__ import annotations

from dataclasses import dataclass, field
from typing import TYPE_CHECKING, Any

import polars as pl
from rich.console import Console
from rich.panel import Panel
from rich.table import Table

if TYPE_CHECKING:
    import pandas as pd


class ResultPlotter:
    

    def __init__(self, result: ConvergenceResult | GroupedConvergenceResult) -> None:
        self.result = result

    def evolution(
        self,
        *,
        show: bool = False,
    ) -> Any:
        
        # 1. Guard against direct plotting on a GroupedConvergenceResult
        if isinstance(self.result, GroupedConvergenceResult):
            available_groups = [dict(k) for k in self.result]
            raise TypeError(
                "Direct evolution plotting on a GroupedConvergenceResult is not supported without selecting a specific group. "
                "Please index a specific group first, for example: "
                "result[key].plot.evolution()\n"
                f"Available groups: {available_groups}"
            )

        # 2. Check matplotlib dependency
        try:
            import matplotlib.pyplot as plt
        except ImportError as exc:
            raise ImportError(
                "Plotting requires matplotlib. Install it via 'uv pip install mocaco[plot]'."
            ) from exc

        df = self.result.data

        # Ensure required columns exist
        missing = [
            col
            for col in (
                self.result.estimate_col,
                self.result.error_col,
                self.result.n_col,
                self.result.is_converged_col,
            )
            if col not in self.result.data.columns
        ]
        if missing:
            raise ValueError(f"Columns not found in diagnostic data for plotting: {missing}")

        # Extract arrays
        n_vals = df.get_column(self.result.n_col).to_numpy()
        estimate_vals = df.get_column(self.result.estimate_col).to_numpy()
        error_vals = df.get_column(self.result.error_col).to_numpy()

        fig, ax = plt.subplots(figsize=(8, 5))

        # Plot estimate line
        ax.plot(n_vals, estimate_vals, label="Estimate", color="#1f77b4", linewidth=2)

        # Plot error band
        ax.fill_between(
            n_vals,
            estimate_vals - error_vals,
            estimate_vals + error_vals,
            color="#1f77b4",
            alpha=0.2,
            label="Error Band",
        )

        # Plot vertical line for convergence (first True occurrence)
        converged_df = df.filter(pl.col(self.result.is_converged_col))
        if not converged_df.is_empty():
            first_converged_n = converged_df.get_column(self.result.n_col)[0]
            ax.axvline(
                x=first_converged_n,
                color="red",
                linestyle="--",
                label=f"Converged at n={first_converged_n}",
            )

        ax.set_xlabel("Number of samples (n)")
        ax.set_ylabel("Estimate")
        ax.set_title(f"Convergence Evolution ({self.result.method})")
        ax.legend()
        ax.grid(True, linestyle=":", alpha=0.7)

        if show:
            plt.show()
            plt.close(fig)

        return ax

    


@dataclass(frozen=True)
class ConvergenceResult:
    

    # Common outputs to all criterions
    method: str
    n: int
    n_col: str
    estimate: float | None
    estimate_col: str
    error: float | None
    error_col: str
    is_converged: bool | None
    is_converged_col: str

    # Build by Wrapper or by the criterion default
    data: pl.DataFrame

    # Performance
    execution_time_sec: float | None = None

    # Method-specific outputs
    diagnostics: dict[str, Any] = field(default_factory=dict)

    @property
    def plot(self) -> ResultPlotter:
        
        return ResultPlotter(self)

    def __getitem__(self, key: Any) -> ConvergenceResult:
        
        raise TypeError(
            f"'{type(self).__name__}' object is not subscriptable. "
            "Group-based indexing is only available when 'group_cols' is specified."
        )

    def to_polars(self) -> pl.DataFrame:
        
        return self.data

    def to_pandas(self) -> pd.DataFrame:
        
        return self.data.to_pandas()

    def summary(self) -> None:
        
        console = Console()

        # Define status colors and icons
        status_color = "green" if self.is_converged else "yellow"
        status_icon = "✔" if self.is_converged else "⚠"
        status_text = "Converged" if self.is_converged else "Not Converged"

        # Create the inner layout table
        table = Table(show_header=False, box=None, padding=(0, 2))
        table.add_row("[bold]Method:[/bold]", f"[cyan]{self.method}[/cyan]")
        table.add_row(
            "[bold]Status:[/bold]", f"[{status_color}]{status_icon} {status_text}[/{status_color}]"
        )
        table.add_row("[bold]Samples (n):[/bold]", str(self.n))
        est_str = f"{self.estimate:.6g}" if self.estimate is not None else "-"
        table.add_row("[bold]Estimate:[/bold]", est_str)
        err_str = f"{self.error:.6g}" if self.error is not None else "-"
        table.add_row("[bold]Error:[/bold]", err_str)
        if self.execution_time_sec is not None:
            table.add_row(
                "[bold]Execution Time:[/bold]", f"{self.execution_time_sec * 1000:.2f} ms"
            )

        # Append diagnostics dynamically if they exist
        if self.diagnostics:
            table.add_row("", "")  # Spacer
            table.add_row("[bold italic]Diagnostics:[/bold italic]", "")
            for key, value in self.diagnostics.items():
                # Format floats cleanly, otherwise just cast to str
                val_str = f"{value:.6g}" if isinstance(value, float) else str(value)
                table.add_row(f"  {key}:", val_str)

        # Wrap in a stylized panel
        panel = Panel(
            table,
            title="[bold]Convergence Summary[/bold]",
            title_align="left",
            border_style=status_color,
            expand=False,
        )

        console.print()
        console.print(panel)
        console.print()

    def __str__(self) -> str:
        
        return str(self.data)

    def __repr__(self) -> str:
        
        return (
            f"ConvergenceResult("
            f"method={self.method!r}, "
            f"n={self.n!r}"
            f"estimate={self.estimate!r}, "
            f"error={self.error!r}, "
            f"is_converged={self.is_converged!r}, "
            f")"
        )

    def error_by(self, *, keep: str | list[str]) -> dict[tuple[tuple[str, Any], ...], float | None]:
        
        raise TypeError(
            "Cannot access 'error_by' on a ConvergenceResult without grouping. "
            "Group-based error aggregation is only available when 'group_cols' is specified. "
            "Please index a specific group first, for example: result[group_key].error_by(...)"
        )


class GroupedConvergenceResult(dict[tuple[tuple[str, Any], ...], ConvergenceResult]):
    

    @property
    def plot(self) -> ResultPlotter:
        
        return ResultPlotter(self)

    @property
    def n(self) -> int:
        
        return sum(res.n for res in self.values())

    @property
    def error(self) -> float | None:
        
        errors = [abs(res.error) for res in self.values() if res.error is not None]
        return sum(errors) if errors else None

    def error_by(self, *, keep: str | list[str]) -> dict[tuple[tuple[str, Any], ...], float | None]:
        
        keep_cols = [keep] if isinstance(keep, str) else list(keep)
        available = {col for group_key in self for col, _ in group_key}
        unknown = [col for col in keep_cols if col not in available]
        if unknown:
            raise ValueError(
                f"Unknown group column(s) {unknown} for 'keep'. "
                f"Available group columns: {sorted(available)}."
            )

        aggregated: dict[tuple[tuple[str, Any], ...], list[float]] = {}
        for group_key, res in self.items():
            sub_key: tuple[tuple[str, Any], ...] = tuple(
                (col, value) for col, value in group_key if col in keep_cols
            )
            bucket = aggregated.setdefault(sub_key, [])
            if res.error is not None:
                bucket.append(abs(res.error))

        return {key: (sum(errors) if errors else None) for key, errors in aggregated.items()}

    @property
    def estimate(self) -> float | None:
        
        raise TypeError(
            "Cannot access 'estimate' directly on a GroupedConvergenceResult. "
            "You must index a specific group first, for example: result[group_key].estimate"
        )

    @property
    def diagnostics(self) -> dict[str, Any]:
        
        raise TypeError(
            "Cannot access 'diagnostics' directly on a GroupedConvergenceResult. "
            "You must index a specific group first, for example: result[group_key].diagnostics"
        )

    @property
    def is_converged(self) -> bool:
        
        return all(res.is_converged for res in self.values())

    @property
    def data(self) -> pl.DataFrame:
        
        dfs = []
        for group_key, res in self.items():
            # Attach group column key-values directly to the group DataFrame
            group_attrs = dict(group_key)
            df_group = res.data.with_columns(
                [pl.lit(val).alias(col) for col, val in group_attrs.items()]
            )
            dfs.append(df_group)

        return pl.concat(dfs) if dfs else pl.DataFrame()

    @property
    def execution_time_sec(self) -> float | None:
        
        times = [
            res.execution_time_sec for res in self.values() if res.execution_time_sec is not None
        ]
        return sum(times) if times else None

    def to_polars(self) -> pl.DataFrame:
        
        return self.data

    def to_pandas(self) -> pd.DataFrame:
        
        return self.data.to_pandas()

    def summary(self) -> None:
        
        for group_key, res in self.items():
            group_str = ", ".join(f"{col}={val}" for col, val in group_key)
            print(f"\n--- Group ({group_str}) ---")
            res.summary()

# Mandatory basic modules
from __future__ import annotations

import warnings
from typing import TYPE_CHECKING

# Basic modules used to build the criterion (According contributor needs)
import polars as pl
from pydantic import BaseModel, Field
from scipy.stats import norm

# Modules needed from mocaco
from mocaco.registry import registry
from mocaco.result import ConvergenceResult, GroupedConvergenceResult

if TYPE_CHECKING:
    from mocaco.protocols import SampleFrame


# 1. Criterion input parameters
class InputParams(BaseModel):
    threshold: float = Field(
        ...,
        gt=0,
        description="Maximum accepted absolute Monte Carlo error.",
    )

    confidence_level: float = Field(
        0.95,
        gt=0,
        lt=1,
        description="Confidence level used for the CLT margin of error.",
    )

    eval_frequency: int | None = Field(
        default=None,
        gt=0,
        description="Step interval for periodic evaluation of convergence history natively in Polars.",
    )


# 2. Criterion logic (calculous)
@registry.register()
class CLTAbsoluteCriterion:
    name = "clt_uni_abs"

    description = (
        "Central Limit Theorem based convergence criterion "
        "using an absolute Monte Carlo error threshold."
    )

    params_type = InputParams

    assumptions = (
        "Samples are independent and identically distributed (i.i.d.).",
        "The target quantity has finite variance.",
        "The sample size is sufficiently large (>30) for the CLT to provide a normal approximation.",
    )

    limitations = (
        "This is a probabilistic stopping criterion, not a proof of convergence.",
        "For small sample sizes, the normal approximation may be inaccurate.",
        "Does not detect systematic bias or non-stationarity in the sampling process.",
        "The stability window introduces a lag in detecting convergence.",
    )

    result_interpretation = (
        "is_converged is True when the CLT-based absolute error has stayed below "
        "the threshold for at least stb_window consecutive iterations. "
        "estimate is the cumulative mean at the final sample. "
        "error is the CLT-based absolute error (z_score * SEM) at the final sample. "
        "n is the total number of samples evaluated."

        ...,
        gt=0,

    )

    confidence_level: float = Field(

@registry.register()
class CLTRelativeCriterion:
    name = "clt_uni_rel"

    description = (
        "Central Limit Theorem based convergence criterion "
        "using a relative Monte Carlo error threshold."
    )

    params_type = InputParams

    assumptions = (
        "Samples are independent and identically distributed (i.i.d.).",
        "The target quantity has finite variance.",
        "The sample size is sufficiently large (>30) for the CLT to provide a normal approximation.",
    )

    limitations = (
        "This is a probabilistic stopping criterion, not a proof of convergence.",
        "For small sample sizes, the normal approximation may be inaccurate.",
        "Does not detect systematic bias or non-stationarity in the sampling process.",
        "The stability window introduces a lag in detecting convergence.",
        "When the estimated value is near zero, the criterion falls back to an absolute error check.",
    )

    result_interpretation = (
        "is_converged is True when the CLT-based relative error has stayed below "
        "the threshold for at least stb_window consecutive iterations. "
        "estimate is the cumulative mean at the final sample. "
        "error is the CLT-based relative error (z_score * SEM / |estimate|) at the final sample. "
        "n is the total number of samples evaluated."
    )

    example_usage = 

    references = ("https://en.wikipedia.org/wiki/Central_limit_theorem",)

    def run(
        self,
        samples: SampleFrame,
        params: InputParams,
    ) -> ConvergenceResult | GroupedConvergenceResult:

        z_score = float(norm.ppf(1.0 - (1.0 - params.confidence_level) / 2.0))

        eps = 1e-10

        def _evaluate_partition(df: pl.DataFrame) -> ConvergenceResult:
            # Assumption
            n_samples = df.height
            if n_samples < 30:
                warnings.warn(
                    f"Sample size n={n_samples} is less than the required minimum of 30 "
                    "for the Central Limit Theorem normal approximation to hold.",
                    UserWarning,
                    stacklevel=2,
                )
                return ConvergenceResult(
                    method=self.name,
                    n=n_samples,
                    n_col="n",
                    estimate=None,
                    estimate_col="mean",
                    error=None,
                    error_col="rel_error",
                    is_converged=False,
                    is_converged_col="is_converged",
                    data=pl.DataFrame(),
                    diagnostics={
                        "threshold": params.threshold,
                        "confidence_level": params.confidence_level,
                        "warning": f"Insufficient sample size (n={n_samples} < 30).",
                    },
                )

            target = pl.col(samples.target_col).cast(pl.Float64)

            # Single-pass evaluation if eval_frequency is not set
            if params.eval_frequency is None or params.eval_frequency == 1:
                agg_df = df.select(
                    [pl.len().alias("n"), target.mean().alias("mean"), target.std().alias("std")]
                )

                final = agg_df.row(0, named=True)
                n = final["n"]
                mean = final["mean"]
                std = final["std"] if final["std"] is not None else 0.0

                sem = std / (n**0.5) if n > 0 else 0.0
                abs_error = sem * z_score

                # Relative error: divide by |estimate|; fall back to absolute when near zero
                if abs(mean) < eps:
                    rel_error = None
                    is_converged = abs_error <= params.threshold
                else:
                    rel_error = abs_error / abs(mean)
                    is_converged = rel_error <= params.threshold

                final_df = pl.DataFrame(

                        "n": [n],
                        "mean": [mean],

                    data=final_df,
